def binomial_recursivo(n,p):
    U = np.random.uniform()
    i = 0 
    pi = (1-p)**n
    F = pi
    while U > F:
        pi = (n - i) / (i + 1) * (p / (1 - p)) * pi
        i = i + 1
        F = F + pi
    return i
